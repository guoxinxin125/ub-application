 for attr in total used available_cleared available_uncleared; do
    attr_path="/sys/kernel/obmm_mempool/obmm-1/$attr"
    if [ -r "$attr_path" ]; then
      printf '%-24s %s\n' "$attr" "$(cat "$attr_path")"
    else
      printf '%-24s %s\n' "$attr" "<missing>"
    fi
  done



  watch -n 0.5 "
  echo '===== Linux hugetlb pool ====='
  grep . \
    $HUGEPAGE_DIR/nr_hugepages \
    $HUGEPAGE_DIR/free_hugepages \
    2>/dev/null

  echo '===== OBMM pool ====='
  grep . \
    $OBMM_POOL_DIR/total \
    $OBMM_POOL_DIR/used \
    $OBMM_POOL_DIR/available_cleared \
    $OBMM_POOL_DIR/available_uncleared \
    2>/dev/null
  "



  cat /sys/devices/system/node/online
  cat /sys/devices/system/node/has_memory
  cat /sys/devices/system/node/has_cpu

  for node_id in 1 3; do
    echo "===== node $node_id ====="
    grep -E 'MemTotal|MemFree' \
      /sys/devices/system/node/node${node_id}/meminfo
  done
